import processing.serial.*;

Serial myPort;
float yaw, pitch, roll;

void setup() {
  size(800, 800, P3D); // Use 3D renderer
  
  // List all serial ports and print them to the console
  printArray(Serial.list());
  
  // NOTE: If the box doesn't move, change [0] to the index of your Arduino's port
  String portName = Serial.list()[0]; 
  myPort = new Serial(this, portName, 115200);
  myPort.bufferUntil('\n');
}

void draw() {
  background(30);
  lights();
  
  pushMatrix();
    translate(width/2, height/2, 0);
    
    // Rotate the 3D view
    rotateX(radians(-pitch));
    rotateZ(radians(roll));
    rotateY(radians(-yaw));

    // Draw a "smartphone-like" box
    stroke(255);
    fill(0, 100, 200);
    box(300, 30, 150); 
    
    // Draw a small red "front" indicator
    fill(255, 0, 0);
    translate(0, 0, 75);
    box(50, 10, 5);
  popMatrix();
  
  // Display Angles
  fill(255);
  textSize(24);
  text("Yaw: " + nfc(yaw, 1) + "°", 50, 50);
  text("Pitch: " + nfc(pitch, 1) + "°", 50, 80);
  text("Roll: " + nfc(roll, 1) + "°", 50, 110);
}

void serialEvent(Serial myPort) {
  String input = myPort.readStringUntil('\n');
  if (input != null) {
    input = trim(input);
    String[] values = split(input, '/');
    if (values.length == 3) {
      yaw = float(values[0]);
      pitch = float(values[1]);
      roll = float(values[2]);
    }
  }
}
