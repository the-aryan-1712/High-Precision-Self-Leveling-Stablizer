#include "Wire.h"
#include <MPU6050_light.h>
#include <Servo.h>

MPU6050 mpu(Wire);
Servo myServo1;
Servo myServo2;
Servo myServo3;
void setup() {
  Serial.begin(9600); 
  Wire.begin();

  // Initialize the sensor
  byte status = mpu.begin();
  if (status != 0) {
    Serial.println("MPU6050 connection failed! Check wiring.");
    while (1) {} // Halt if sensor isn't found
  }

  myServo1.attach(8);
  myServo2.attach(9);
  myServo3.attach(10);

  // Calibrate the sensor. Keep it perfectly flat and still!
  Serial.println("Calibrating...");
  delay(1000);
  mpu.calcOffsets(true, true); 
  Serial.println("Calibration complete.");
  delay(1000);
}

void loop() {
  mpu.update();

  // Get Yaw, Pitch, and Roll
  float yaw = -mpu.getAngleZ();
  float pitch = -mpu.getAngleY();
  float roll = mpu.getAngleX();

  // Send data in the format: yaw/pitch/roll
  Serial.print(yaw);
  Serial.print("/");
  Serial.print(pitch);
  Serial.print("/");
  Serial.print(roll); 


  int rollAngle  = map(roll,  -90, 90, 0, 180);
  int pitchAngle = map(pitch, -90, 90, 0, 180);
  int yawAngle   = map(yaw,    -90, 90, 0, 180);

  rollAngle  = constrain(rollAngle, 0, 180);
  pitchAngle = constrain(pitchAngle, 0, 180);
  yawAngle   = constrain(yawAngle, 0, 180);

  // Move servos
  myServo2.write(rollAngle);
  myServo1.write(pitchAngle);
  myServo3.write(yawAngle);

  // int servoAngle = map(pitch, -90, 90, 0, 180);
  // servoAngle = constrain(servoAngle, 0, 180);

  // myServo.write(constrain(servoAngle, 0, 180));
  // myServo.write(servoAngle);
  // myServo.write(servoAngle);


  // Serial.print("/");
  // Serial.println(servoAngle); 

  Serial.print("R: "); Serial.print(roll);
  Serial.print(" P: "); Serial.print(pitch);
  Serial.print(" Y: "); Serial.println(yaw);

  // delay(100); // Run at ~50fps
}
